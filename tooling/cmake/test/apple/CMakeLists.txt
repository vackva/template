cmake_minimum_required(VERSION 3.18)
project(tanh_apple_test CXX)
enable_testing()
include(${CMAKE_CURRENT_LIST_DIR}/../../apple.cmake)

tanh_apple_deployment_target(MACOS 12.0 IOS 14.0)
tanh_apple_sysroot_from_xcrun()
file(WRITE "${CMAKE_BINARY_DIR}/t.cpp" "int main() { return 0; }\n")
add_executable(apple_probe "${CMAKE_BINARY_DIR}/t.cpp")
tanh_ios_disable_code_signing(apple_probe)
tanh_ios_test_bundle(apple_probe BUNDLE_ID_PREFIX com.tanh.test)
if(APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    if(NOT CMAKE_OSX_DEPLOYMENT_TARGET)
        message(FATAL_ERROR "CMAKE_OSX_DEPLOYMENT_TARGET not defaulted")
    endif()
    if(NOT CMAKE_OSX_SYSROOT)
        message(FATAL_ERROR "CMAKE_OSX_SYSROOT not filled in")
    endif()
elseif(NOT APPLE)
    get_target_property(_b apple_probe MACOSX_BUNDLE)
    if(_b)
        message(FATAL_ERROR "tanh_ios_test_bundle must be a no-op off iOS")
    endif()
endif()
add_test(NAME apple_probe COMMAND apple_probe)
