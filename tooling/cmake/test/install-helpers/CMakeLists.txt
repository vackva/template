cmake_minimum_required(VERSION 3.18)
project(tanh_install_helpers_test CXX)
enable_testing()
include(${CMAKE_CURRENT_LIST_DIR}/../../install-helpers.cmake)

file(WRITE "${CMAKE_BINARY_DIR}/lib.cpp" "int f() { return 1; }\n")
add_library(rp_lib SHARED "${CMAKE_BINARY_DIR}/lib.cpp")
add_library(rp_lib2 SHARED "${CMAKE_BINARY_DIR}/lib.cpp")
tanh_set_install_rpath(rp_lib rp_lib2 EXTRA_ELF_PATHS /opt/x/lib EXTRA_APPLE_PATHS /opt/y/lib)
get_target_property(_rp rp_lib2 INSTALL_RPATH)
if(TANH_BINARY_FORMAT STREQUAL "ELF" AND NOT "${_rp}" STREQUAL "$ORIGIN;/opt/x/lib")
    message(FATAL_ERROR "INSTALL_RPATH = '${_rp}'")
elseif(TANH_BINARY_FORMAT STREQUAL "Mach-O" AND NOT "${_rp}" STREQUAL "@loader_path;/opt/y/lib")
    message(FATAL_ERROR "INSTALL_RPATH = '${_rp}'")
endif()
add_test(NAME install-helpers-configured COMMAND ${CMAKE_COMMAND} -E true)
