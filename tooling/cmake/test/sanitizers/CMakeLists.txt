cmake_minimum_required(VERSION 3.18)
project(tanh_sanitizers_test CXX)
enable_testing()
include(${CMAKE_CURRENT_LIST_DIR}/../../sanitizers.cmake)

file(WRITE "${CMAKE_BINARY_DIR}/lib.cpp" "int f() { return 1; }\n")
add_library(san_lib EXCLUDE_FROM_ALL "${CMAKE_BINARY_DIR}/lib.cpp")  # not built: flags only
if(MSVC)
    tanh_add_sanitizer(san_lib asan DEFINE TEST_WITH_ASAN)
    set(_expect "/fsanitize=address")
else()
    file(WRITE "${CMAKE_BINARY_DIR}/ignore.txt" "src:*/vendored/*\n")
    tanh_add_sanitizer(san_lib ubsan DEFINE TEST_WITH_UBSAN IGNORELIST "${CMAKE_BINARY_DIR}/ignore.txt")
    set(_expect "-fsanitize=undefined")
endif()
get_target_property(_opts san_lib INTERFACE_COMPILE_OPTIONS)
get_target_property(_defs san_lib INTERFACE_COMPILE_DEFINITIONS)
if(NOT "${_opts}" MATCHES "${_expect}")
    message(FATAL_ERROR "sanitizer flag missing from INTERFACE_COMPILE_OPTIONS: ${_opts}")
endif()
if(NOT "${_defs}" MATCHES "TEST_WITH_")
    message(FATAL_ERROR "sanitizer define missing from INTERFACE_COMPILE_DEFINITIONS: ${_defs}")
endif()
if(NOT MSVC)
    get_target_property(_lopts san_lib INTERFACE_LINK_OPTIONS)
    if(NOT "${_lopts}" MATCHES "${_expect}")
        message(FATAL_ERROR "sanitizer flag missing from INTERFACE_LINK_OPTIONS: ${_lopts}")
    endif()
    if(NOT "${_opts}" MATCHES "fsanitize-ignorelist=")
        message(FATAL_ERROR "ignorelist missing: ${_opts}")
    endif()
endif()
add_test(NAME sanitizers-configured COMMAND ${CMAKE_COMMAND} -E true)
